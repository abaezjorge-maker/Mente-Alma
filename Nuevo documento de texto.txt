    let currentVerse = null;

    function generateVerse(random = false) {
      const nameInput = document.getElementById("name");
      const name = nameInput.value.trim();

      if (name === "") {
        alert("Por favor, escribe tu nombre primero.");
        nameInput.focus();
        return;
      }

      let index;

      if (random) {
        index = Math.floor(Math.random() * verses.length);
      } else {
        const today = new Date();
        const dateCode = today.getFullYear() + today.getMonth() + today.getDate();
        let nameCode = 0;

        for (let i = 0; i < name.length; i++) {
          nameCode += name.charCodeAt(i);
        }

        index = (nameCode + dateCode) % verses.length;
      }

      currentVerse = verses[index];

      document.getElementById("greeting").textContent =
        "Hola, " + name + " 👋 Este versículo es para ti hoy:";

      document.getElementById("verse").textContent =
        "“" + currentVerse.text + "”";

      document.getElementById("reference").textContent =
        currentVerse.reference;

      document.getElementById("reflection").textContent =
        "💭 Reflexión: " + currentVerse.reflection;

      document.getElementById("result").style.display = "block";
    }

    function copyVerse() {
      const name = document.getElementById("name").value.trim();

      if (!currentVerse) return;

      const textToCopy =
        "Hola, " + name + ". Este versículo es para ti hoy:\n\n" +
        "“" + currentVerse.text + "”\n" +
        currentVerse.reference + "\n\n" +
        "Reflexión: " + currentVerse.reflection + "\n\n" +
        "Mente & Alma 🧠✨";

      navigator.clipboard.writeText(textToCopy)
        .then(() => {
          alert("Versículo copiado correctamente.");
        })
        .catch(() => {
          alert("No se pudo copiar el texto.");
        });
    }
